<script>
const bids = [
  {prize:"R$ 1.000,00", bid:"R$ 3,00",  when:"Hoje, 09:41", state:"live"},
  {prize:"R$ 500,00",   bid:"R$ 12,00", when:"12/10, 14:32", state:"lost"},
  {prize:"R$ 100,00",   bid:"R$ 4,50",  when:"11/10, 20:15", state:"won"},
  {prize:"R$ 250,00",   bid:"R$ 8,20",  when:"09/10, 18:47", state:"paid"}
];

function action(b, i){
  if (b.state === "live") return '<span class="status live">Você está ganhando</span>';
  if (b.state === "lost") return '<span class="status">Cobriram seu lance</span>';
  if (b.state === "won")  return '<button class="btn" data-i="'+i+'">Resgatar</button>';
  return '<span class="status paid">Resgatado</span>';
}

function render(){
  document.getElementById("list").innerHTML = bids.map((b,i)=>
    '<li class="bid '+(b.state==="lost"?"lost":"")+'">'+
      '<div class="info">'+
        '<span class="lbl">Prêmio</span>'+
        '<span class="prize">'+b.prize+'</span>'+
        '<div class="mine">'+(b.state==="won"?'<b class="wonlbl">Você ganhou</b> com lance de ':'Seu lance ')+b.bid+'</div>'+
      '</div>'+ action(b,i) +
    '</li>').join("");
}

document.getElementById("list").addEventListener("click", e=>{
  const btn = e.target.closest("button[data-i]");
  if(!btn) return;
  bids[btn.dataset.i].state = "paid";
  render();
});
render();
</script>
