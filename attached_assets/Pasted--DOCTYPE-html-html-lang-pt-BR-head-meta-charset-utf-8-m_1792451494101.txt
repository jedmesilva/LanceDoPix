<script>
const bids = [
  {prize:"R$ 1.000,00", bid:"R$ 157", state:"live"},
  {prize:"R$ 500,00",   bid:"R$ 12",  state:"lost"},
  {prize:"R$ 100,00",   bid:"R$ 45",  state:"won"},
  {prize:"R$ 250,00",   bid:"R$ 82",  state:"paid"}
];
const STATUS = {
  live:["Você está ganhando",""],
  won:["Você ganhou",""],
  lost:["Cobriram seu lance","lost"],
  paid:["Resgatado","paid"]
};

function render(){
  document.getElementById("list").innerHTML = bids.map((b,i)=>{
    const [txt,cls] = STATUS[b.state];
    return '<li class="card '+(b.state==="lost"?"lost":"")+'">'+
      '<div class="row1"><span class="lbl">Prêmio</span><span class="status '+cls+'">'+txt+'</span></div>'+
      '<div class="row2"><span class="prize">'+b.prize+'</span>'+
        (b.state==="won"?'<button class="btn" data-i="'+i+'">Resgatar</button>':'')+
      '</div>'+
      '<div class="mine"><span>'+(b.state==="won"?"Seu lance vencedor":"Seu lance")+'</span><b>'+b.bid+'</b></div>'+
    '</li>';
  }).join("");
}

document.getElementById("list").addEventListener("click", e=>{
  const btn = e.target.closest("button[data-i]");
  if(!btn) return;
  bids[btn.dataset.i].state = "paid";
  render();
});
render();
</script>
